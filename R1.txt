library(limma)

targets <- readTargets("Targets.txt")

RG <- read.maimages(
    targets$FileName,
    source = "agilent"
)

MA <- normalizeWithinArrays(
    RG,
    method = "loess",
    bc.method = "normexp",
    offset = 50
)

dim(RG)
summary(RG$R)

plotMD(RG, column = 2)
plotMA3by2(MA)

MA.q <- normalizeBetweenArrays(
    MA,
    method = "quantile"
)

plotDensities(RG)
plotDensities(MA)

design <- modelMatrix(
    targets,
    ref = "normal"
)

fit <- lmFit(MA.q, design)

eb <- eBayes(fit)

tt <- topTable(
    eb,
    n = nrow(MA.q),
    adjust.method = "BH"
)

top50 <- head(tt, 50)
