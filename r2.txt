# SESSIONAL Q3: MICROARRAY
# TOP 50 DEGs + CLUSTERING
# ================================

# 1. Load package
library(limma)

# 2. Read target file
targets <- readTargets("Targets.txt")

# 3. Read microarray data
RG <- read.maimages(
  targets$FileName,
  source = "agilent"
)

# 4. Within-array normalization
MA <- normalizeWithinArrays(
  RG,
  method = "loess",
  bc.method = "normexp",
  offset = 50
)

# 5. Between-array normalization
MA.q <- normalizeBetweenArrays(
  MA,
  method = "quantile"
)

# 6. Design matrix
design <- modelMatrix(
  targets,
  ref = "normal"
)

# 7. Linear model
fit <- lmFit(
  MA.q,
  design
)

# 8. Empirical Bayes
eb <- eBayes(fit)

# 9. Differential expression table
tt <- topTable(
  eb,
  n = nrow(MA.q),
  adjust.method = "BH"
)

# 10. Select top 50 DEGs
top50 <- head(tt, 50)

# View top 50
top50

# 11. Extract expression values of top 50
top50_expr <- MA.q[rownames(top50), ]

# Check dimensions
dim(top50_expr)

# =================================
# A. HIERARCHICAL CLUSTERING
# =================================

heatmap(
  as.matrix(top50_expr),
  scale = "row",
  main = "Hierarchical Clustering of Top 50 DEGs"
)

# Save hierarchical clustering
png(
  "Top50_Hierarchical_Clustering.png",
  width = 1000,
  height = 800
)

heatmap(
  as.matrix(top50_expr),
  scale = "row",
  main = "Hierarchical Clustering of Top 50 DEGs"
)

dev.off()

# =================================
# B. K-MEANS CLUSTERING
# =================================

# Scale expression values
scaled_data <- scale(top50_expr)

# K-means
set.seed(123)

km <- kmeans(
  scaled_data,
  centers = 3
)

# View cluster assignment
km$cluster

# Order genes according to K-means cluster
km_ordered <- top50_expr[
  order(km$cluster),
]

# Plot K-means clusters
heatmap(
  as.matrix(km_ordered),
  scale = "row",
  main = "K-means Clustering of Top 50 DEGs"
)

# Save K-means clustering
png(
  "Top50_Kmeans_Clustering.png",
  width = 1000,
  height = 800
)

heatmap(
  as.matrix(km_ordered),
  scale = "row",
  main = "K-means Clustering of Top 50 DEGs"
)

dev.off()
